# Native-repo packages for Debian/Ubuntu-family Linux (PikaOS, etc.), installed
# by install.sh via `apt install`. One package per line, '#' comments allowed.
#
# This is a starting core set, not a full mirror of Brewfile — package names
# and availability vary by distro/release, so verify with `apt-cache search`
# on the actual box before assuming something here is right. Tools without a
# reliable native package (neovim, pyenv, nvm, topgrade, lazygit, oh-my-tmux,
# zsh) are installed separately by install.sh via their own upstream
# installers.
#
# Desktop/GUI apps live in apt-packages-desktop.txt instead, so a headless box
# doesn't get them.

build-essential
cmake
git
curl
wget
tmux
lf

# Needed by install.sh itself before any repo can be added (add_apt_repo runs
# `curl | gpg` over HTTPS). install.sh installs these first regardless; they're
# listed here so re-runs keep them.
gnupg
ca-certificates

# Needed by things this repo actually ships:
# unzip            — install.sh's Nerd Font step
# command-not-found — .zshrc enables the oh-my-zsh plugin, which sources
#                     /etc/zsh_command_not_found from this package
# cifs-utils       — mount.cifs, for a `synology`-style alias in
#                    ~/.zsh_aliases.local
# fontconfig       — fc-cache, after the font install
unzip
command-not-found
cifs-utils
fontconfig

libreadline-dev

# psmisc provides fuser, which install.sh/bootstrap.sh rely on to detect
# dpkg-lock contention for the apt retry loop (issue #163). Without it the
# retry logic silently degrades to a single attempt.
psmisc
